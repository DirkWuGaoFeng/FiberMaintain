│  │      // 对端有源盘列表 (场景2)                                              │    │
│  │      vector<PeerInfo> peers;                                                │    │
│  │      // PeerInfo: {board_id, port_id, via_port, role(PORT2/PORT3)}         │    │
│  │                                                                             │    │
│  │      // 主路标识 (场景2: Port-2对端为主路)                                  │    │
│  │      optional<PeerInfo> primary_peer;   // 主路 (Port-2对端)                │    │
│  │      optional<PeerInfo> backup_peer;    // 备路 (Port-3对端, 仅Case A)      │    │
│  │  };                                                                         │    │
│  └─────────────────────────────────────────────────────────────────────────────┘    │
│                                                                                     │
│  resolve() 内部流程:                                                                │
│  ┌─────────────────────────────────────────────────────────────────────────────┐    │
│  │  1. 从 fiber_by_id_ 获取 FiberCacheEntry                                   │    │
│  │  2. 判断 dst_board_type:                                                    │    │
│  │     · ACTIVE → scene=1, case=0, 无peers                                    │    │
│  │     · PASSIVE → scene=2, 继续查找peers                                     │    │
│  │  3. 若 PASSIVE:                                                             │    │
│  │     · 查 fiber_by_port_[{dst_board, 2}] → 有网元内连纤?                    │    │
│  │       · 有 → 获取对端有源盘 → primary_peer (Port-2)                        │    │
│  │       · 无 → case=C, 返回                                                  │    │
│  │     · 查 fiber_by_port_[{dst_board, 3}] → 有网元内连纤?                    │    │
│  │       · 有 → 获取对端有源盘 → backup_peer (Port-3)                         │    │
│  │       · 无 → case=B                                                        │    │
│  │     · 两者都有 → case=A                                                    │    │
│  └─────────────────────────────────────────────────────────────────────────────┘    │
└─────────────────────────────────────────────────────────────────────────────────────┘
```

│  │  Step 1: Layer 1 — 批量拓扑解析 (单次读锁)                                 │    │
│  │                                                                             │    │
│  │  vector<FiberTopologyInfo> topos =                                          │    │
│  │      resolver_.resolve_batch(request->fiber_ids());                         │    │
│  │                                                                             │    │
│  │  内部:                                                                      │    │
│  │    shared_lock<shared_mutex> lock(cache_rw_mutex_);  // 一次读锁           │    │
│  │    for (auto& fid : fiber_ids)                                              │    │
│  │        topos.push_back(resolve_internal(fid));                              │    │
│  │    → 释放读锁                                                               │    │
│  │                                                                             │    │
│  │  ⚠️ 百万级: 若 fiber_ids > 10000, 分片处理 (每片5000)                     │    │
│  └─────────────────────────────────────────────────────────────────────────────┘    │
│    │                                                                                │
│    ▼                                                                                │
│  ┌─────────────────────────────────────────────────────────────────────────────┐    │
│  │  Step 2: Layer 2 — 批量构建 + 去重                                          │    │
│  │                                                                             │    │
│  │  DeduplicatedPerfTargets dedup = perf_builder_.build_batch(topos);          │    │
│  │                                                                             │    │
│  │  内部:                                                                      │    │
│  │    set<PortKey> unique_ports;                                               │    │
│  │    vector<PerfTarget> all_targets;                                          │    │
│  │    for (auto& topo : topos)                                                 │    │
│  │        auto targets = build(topo);  // 场景2仅主路                          │    │
│  │        for (auto& t : targets)                                              │    │
│  │            unique_ports.insert(t.port);                                     │    │
│  │            all_targets.push_back(t);                                        │    │
│  │                                                                             │    │
│  │  效果: 1000条连纤 → 可能仅500个唯一端口 (去重50%)                           │    │
│  └─────────────────────────────────────────────────────────────────────────────┘    │
│    │                                                                                │
│    ▼                                                                                │
│  ┌─────────────────────────────────────────────────────────────────────────────┐    │
│  │  Step 3: Layer 3 — 单次RPC (或分片RPC)                                     │    │
│  │                                                                             │    │
│  │  vector<PerfResult> results =                                               │    │
│  │      perf_executor_.batch_execute(topos, perf_stub_);                       │    │
│  │                                                                             │    │
│  │  内部:                                                                      │    │
│  │    if (unique_ports.size() <= 500)                                          │    │
│  │        → 单次 BatchGetCurrentPerformance RPC                                │    │
│  │    else                                                                     │    │
│  │        → 分片: 每片500端口 → 多次RPC → 合并结果                            │    │
│  │                                                                             │    │
│  │    → 解析响应 → 按 fiber_id 映射到各连纤                                   │    │
│  └─────────────────────────────────────────────────────────────────────────────┘    │
│    │                                                                                │
│    ▼                                                                                │
│  ┌─────────────────────────────────────────────────────────────────────────────┐    │
│  │  Step 4: 构造批量响应                                                       │    │
│  │                                                                             │    │
│  │  for (auto& result : results)                                               │    │
│  │      → 填充 response->add_results()                                        │    │
│  └─────────────────────────────────────────────────────────────────────────────┘    │
│                                                                                     │
│  性能对比:                                                                          │
│  ┌─────────────────────────────────────────────────────────────────────────────┐    │
│  │  优化前: 1000条连纤 × 2次RPC = 2000次RPC                                  │    │

## 5. 异步启动与Pull Call机制

### 5.1 Pull Call 机制说明

```
┌─────────────────────────────────────────────────────────────────────────────────────┐
│  Pull Call 拉模式机制                                                               │
│                                                                                     │
│  传统同步模式 (❌ 问题):                                                            │
│  ┌─────────────────────────────────────────────────────────────────────────────┐    │
│  │  服务启动 → CreatePullCall → 等待全量告警返回 → 写入缓存 → 全量重算        │    │
│  │           → 启动Stream → 启动gRPC Server                                    │    │
│  │                                                                             │    │
│  │  问题: 百万级告警数据量大，等待时间可能超过30s，导致启动超时                 │    │
│  └─────────────────────────────────────────────────────────────────────────────┘    │
│                                                                                     │
│  异步模式 (✅ 优化后):                                                              │
│  ┌─────────────────────────────────────────────────────────────────────────────┐    │
│  │  服务启动 → 立即启动Stream + gRPC Server (服务可用)                         │    │
│  │           → 异步 CreatePullCall → 后台分批接收告警                          │    │
│  │           → 接收完成 → 推入事件队列 → 事件引擎处理全量同步                  │    │
│  │           → 全量重算 (分批) → 完成                                          │    │
│  │                                                                             │    │
│  │  优势: 服务秒级可用，全量同步在后台完成                                     │    │
│  └─────────────────────────────────────────────────────────────────────────────┘    │
└─────────────────────────────────────────────────────────────────────────────────────┘
```

### 5.2 异步启动时序图

```
┌─────────────────────────────────────────────────────────────────────────────────────┐
│                    CoalescingEventQueue 合并示意                                     │
│                                                                                     │
│  时间轴:  T0        T1        T2        T3        T4                                │
│           │         │         │         │         │                                 │

│  │                                                         │                       │
│  │  T0: {301,1,1} → RAISED                                │                       │

│  │                                                         │                       │
│  │  [CREATED F001] → [DELETED F002] → [CREATED F003]      │                       │

│                                                                                     │
│  ─────────────────────────────────────────────────────────────────────               │
│                                                                                     │

│                                                                                     │
